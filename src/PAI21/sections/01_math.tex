\section{Math / Probability}\label{sec:math}

\begin{itemize}
    % Distributions
    \item \textbf{Bernoulli:} (biased) coin flip, $\P{X{=}H}=p$, $\P{X{=}T}=1-p)$, yields \textit{Binomial} distr.
    \item \textbf{Categorical:} (biased) $m$-sided dice, $\P{X{=}i} = p_i$, $p_i \geq 0$, $\sum p_i = 1$, yields \textit{Multinomial} distr.
    \item \textbf{Gaussian:} \highlight{$p(x) = \frac{1}{\sqrt{2\pi\sigma^2}} \exp\paren*{-\frac{(x-\mu)^2}{2\sigma^2}}$}
    % Joint distributions
    \item \textbf{Cond. prob.:} $\P{X\mid Y} = \frac{\P{X\land Y}}{\P{Y}}$ if $\P{Y} \neq 0$
    \item \textbf{Chain rule:} $\P{X\land Y} = \P{X\mid Y} \P{Y}$,\enskip $\P{X_1, {\ldots}, X_n} \equiv \P{X_{1:n}} = \P{X_1} \P{X_2\mid X_1} {\ldots} \P{X_n\mid X_{1:n-1}}$
    \item \textbf{Marg.:} $\P{X_{1:i-1}, X_{i+1:n}} = \sum_{x_i} \P{X_{1:i-1}, x_i, X_{i+1:n}}$,\enskip $\P{X_i} = \sum_{x_1} {\ldots} \sum_{x_n} \P{x_{1:i-1}, X_i, X_{i+1:n}}$
    % Independence
    \item \textbf{$\normalcolor X{\perp}Y$:} $\phantom{\vert Z}\P{X,Y} \overset{\mathrm{chain}}{=} \P{X} \P{Y\mid X} \overset{\mathrm{indep}}{=} \P{X} \P{Y}$
    \item \textbf{$\normalcolor X{\perp}Y\vert Z$:} $\P{x,y\orange\vert z} = \P{x\orange\vert z} \P{y\orange\vert z}$, $\P{X\vert Y\orange,Z} = \P{X\orange\vert Z}$
    % Bayesian inference
    \item \emph{Bayes:} \highlight*{$\P{X\mid Y} = \frac{\P{X,Y}}{\P{Y}}$},\enskip \begin{tabular}{@{}>{\normalcolor\small}l}prior $\P{X}$, likel. $\P{Y\vert X}$,\\ post. $\P{X\vert Y}$, evid. $\P{Y}$\end{tabular}
\end{itemize}

\subsection{Multivariate Gaussian
    \textnormal\hfill$\normalcolor \bm\Sigma = \brace{\sigma_{ij}}$}\label{subsec:mv-gaussian}

\highlight*{$\Gauss{\bm y\mid\bm\mu,\bm\Sigma} = \frac{1}{(2\pi)^{\nicefrac{n}{2}}\sqrt{\abs{\bm\Sigma}}} \exp\paren*{-\frac{1}{2} (\bm y-\bm\mu)^\top \bm\Sigma^{-1} (\bm y-\bm\mu)}$}
\begin{itemize}\vspace{-1pt}
    \item $\sigma_{ij} = \E{(X_i - \mu_i)(X_j - \mu_j)} = \Cov{X_i, X_j} \overset{X_i\perp X_j}{=} 0$
    \item $\sigma_i^2 = \E{(X_i - \mu_i)^2} = \V{X_i}$
\end{itemize}

Gaussians are closed under conditioning, marginalisation, multiplication and addition.

    \item $\P{\bm X_A \vert \bm x_B} = \Gauss{\bm\mu_{A\mid B}, \bm\Sigma_{A\mid B}}$ where $\bm\mu_{A\mid B} = \bm\mu_A + \bm\Sigma_{AB} \bm\Sigma_{BB}^{-1} (\bm x_B - \bm\mu_B)$ and $\bm\Sigma_{A\mid B} = \bm\Sigma_{AA} - \bm\Sigma_{AB} \bm\Sigma_{BB}^{-1} \bm\Sigma_{BA}$
