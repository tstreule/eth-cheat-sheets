    \item \emph{Marginal:} $\P{X_i} = \sum_{x_1} {\ldots} \sum_{x_n} \P{x_{1:i-1}, X_i, X_{i+1:n}}$,\enskip $\P{X_{1:i-1}, X_{i+1:n}} = \sum_{x_i} \P{X_{1:i-1}, x_i, X_{i+1:n}}$

    \item \textbf{$\normalcolor X{\perp}Y$:}\quad~$\P{X,Y} \overset{\mathrm{chain}}{=} \P{X} \P{Y\mid X} \overset{\mathrm{indep}}{=} \P{X} \P{Y}$
    \item[] \textbf{$\normalcolor X{\perp}Y\vert Z$:} $\P{x,y\orange{\vert z}} = \P{x\orange{\vert z}} \P{y\orange{\vert z}}$,\enskip $\P{X\vert Y\orange{,Z}} = \P{X\orange{\vert Z}}$
\end{itemize}

\begin{itemize}\vspace{-.2em}
    \item $\sigma_{ij} = \E{(X_i - \mu_i)(X_j - \mu_j)} = \Cov{X_i, X_j} \overset{X_i\perp X_j}{=} 0$

\end{itemize}
