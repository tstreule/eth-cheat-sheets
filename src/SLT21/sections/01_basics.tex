\section{Basics}

% ===
\begin{itemize}
    \item General p-norm:\enskip
        $\norm{x}_p = \paren{\sum_{i=1}^n \abs{x_i}^p}^{1/p}$
    \item Taylor:\enskip
        $f(x) \approx f(a) + \frac{f'(a)}{1!}(x{-}a) + \frac{f''(a)}{2!}(x{-}a)^2 + ..$
        \begin{itemize}
            \item $f(\symbf x) \approx f(\symbf a) + \pderiv{f(\symbf x)}{\symbf x}\big\vert_{\symbf a} - \frac12 (\symbf x - \symbf a)^\top \frac{\partial^2 f(\symbf x)}{\partial \symbf x \partial \symbf x^\top}\big\vert_{\symbf a} (\symbf x - \symbf a)$
            \vspace{-2pt}
            \item Power series of exp.:\enskip
                $\exp(x) \coloneqq \sum_{k=0}^\infty \frac{x^k}{k!}$
        \end{itemize}
    \item Entropy:\enskip
        $H(X) \equiv H(p_X) = \E[X]{-\log \P{X=x}}$
        \begin{itemize}
            \item $H(X\mid Y) = \sum_y \P{Y=y} H(X\mid Y=y) \leq H(X)$
            \item $H(X,Y) = H(X) + H(Y\mid X)$
            \item $H(X\mid g(X)) \geq 0$
                \quad$\circ$ $H(g(X)\mid X) = 0$
            \item $H(5X) \scriptsize\begin{cases} =H(X) & \text{discrete} \\ >H(X) & \text{continuous} \end{cases}$
        \end{itemize}
    \item MI:\enskip
        $I(X;Y \textcolor{OrangeRed}{\mid Z} ) = H(X \textcolor{OrangeRed}{\mid Z} ) - H(X\mid Y \textcolor{OrangeRed}{, Z} )$ {\scriptsize\enskip(symmetric)}
        \begin{itemize}
            \item $I(X;Y) = D\ped{KL} (p(x,y) \parallel p(x) p(y)) \geq 0$
            \item $I(X_1,\ldots,X_n;Z) = \sum_{i=1}^n I(X_i; Z \mid X_1,\ldots,X_{i-1})$\\
                Markov chain:\enskip $I(X_1;X_2,X_3,\ldots) = I(X_1;X_2)$
            \item $I(X,Y;Z) = I(X;Z) + I(Y;Z\mid X)$
        \end{itemize}
    \item KL-divergence:\enskip
        $D\ped{KL} (p\parallel q) = \sum_x p(x)\log\paren*{\frac{p(x)}{q(x)}} \geq 0$
    \item Cauchy-Schwarz:\enskip
        $\abs{\E{X,Y}}^2 \leq \E{X^2} \E{Y^2}$
    \item $1-z \leq \exp(-z)$
    \item Jensen, $f(X)$ convex:\enskip
        $f(\E{X}) \leq \E{f(X)}$
\end{itemize}

\subsection{Probability / Statistics}

\begin{itemize}
    \item Gaussian:\enskip
        $\Gauss*{x \mid \mu, \sigma^2} = \frac{1}{\sqrt{2\pi\sigma^2}} \exp\paren*{-\frac12 \frac{(x-\mu)^2}{\sigma^2}}$
        \\
        $\Gauss{\symbf x \mid \symbf\mu,\symbf\Sigma} = \frac{1}{\sqrt{(2\pi)^d \abs{\symbf\Sigma}}} \exp \left( -\frac{1}{2} (\symbf x - \mu)^\top \symbf \Sigma^{-1} (\symbf x - \mu) \right)$
        \begin{itemize}
            % \item $\ln(\Gauss{}) = - \frac{\ln\abs{\symbf\Sigma}}{2} - \frac12 (\symbf x - \symbf\mu)^\top \symbf\Sigma^{-1} (\symbf x - \symbf\mu) + \frac{d}{2} \ln(2\pi)$%{-} \frac{d}{2} \ln(2\pi)
            \item $X \sim \Gauss{\symbf{\mu,\Sigma}}, Y = A + BX \implies Y \sim \Gauss{A+B\symbf\mu, B\symbf\Sigma B^\top}$
        \end{itemize}
    \item Binomial:\enskip
        $f(k,n;p) = \P{X=k} = \binom nk p^k (1-p)^{n-k}$
    \item $\V{X} = \E{(X-\E{X})^2} = \E{X^2} - \E{X}^2$
        \\
        $\V{X+Y} = \V{X} + \V{Y} + 2\Cov{X,Y}$
    \item $\Cov{X,Y} = \E{(X-\E{X})(Y-\E{Y})}$\\
        $\phantom{\Cov{X,Y}} = \E{XY} - \E{X}\E{Y}$
        \\
        $\Cov{aX,bY} = ab\Cov{X,Y}$
\end{itemize}

\subsection{Calculus}

\begin{itemize}
    \item Partial:\enskip $\int u v' \diff x = u v - \int u' v \diff x$
        \quad\textbullet\,
        $\pderiv{}{x} \frac{g}{h} = \frac{g'h}{h^2} - \frac{gh'}{h^2}$
    \item $\pderiv{}{\symbf x} (\norm{\symbf x - \symbf b}_2) = \frac{\symbf x-\symbf b}{\norm{\symbf x-\symbf b}_2}$
        \quad\textbullet\,
        $\deriv{}{x} \abs x = \frac{x}{\abs x}$
    \item $\pderiv{}{\symbf X} \log \abs{\symbf X} = \symbf X^{-\top}$
        \quad\textbullet\,
        $\abs{\symbf X^{-1}} = \abs{\symbf X}^{-1}$
        
    \item $\pderiv{}{\symbf x} (\symbf b^\top\symbf  x) = \pderiv{}{\symbf x} (\symbf x^\top\symbf  b) = \symbf b$
    \item $\pderiv{}{\symbf x} (\symbf b^\top \symbf{Ax}) = \symbf A^\top \symbf b$
        \quad\textbullet\,
        $\pderiv{}{\symbf X} (\symbf c^\top \symbf{Xb}) = \symbf{cb}^\top$
    \item $\pderiv{}{\symbf X} (\symbf c^\top \symbf X^\top \symbf b) = \symbf{bc}^\top$
        \quad\textbullet\,
        $\pderiv{}{\symbf x} (\symbf x^\top \symbf x) = 2\symbf x$
    \item $\pderiv{}{\symbf x} (\symbf x^\top \symbf{Ax}) = (\symbf A^\top + \symbf A)\symbf x \overset{\symbf A\textrm{ sym.}}{=} 2\symbf{Ax}$
    \item $\pderiv{}{\symbf X} T\!r(\symbf X^\top \symbf A) = \symbf A$
        \quad\textbullet\,
        Trace trick:\enspace
            $\symbf x^\top \symbf{Ax} = \ldots$\\\hfill
            $\ldots \overset{\small\substack{\textrm{inn. prod.}}}{=} T\!r (\symbf x^\top \symbf{Ax}) \overset{\small\substack{\textrm{cycl. permut.}}}{=} T\!r(\symbf{xx}^\top \symbf A) = T\!r(\symbf{Axx}^\top)$
    \item $\sigma(x) = \frac{1}{1 + \exp(-x)} \implies \nabla\sigma(x) = \sigma(x)(1-\sigma(x))$
    \item $\tanh(x) = \frac{2\sinh(x)}{2\cosh(x)} = \frac{\eu^x - \eu^{-x}}{\eu^x+\eu^{-x}}$
        \hfill\textbullet\hfill
        $\nabla\tanh(x) {=} 1 {-} \tanh^2(x)$
\end{itemize}
